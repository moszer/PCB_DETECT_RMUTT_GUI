## 3.1 ภาพรวมของระบบ

ระบบตรวจจับข้อบกพร่องบนแผ่นวงจรพิมพ์ (PCB) ประกอบด้วย 2 ส่วนหลัก ได้แก่ (1) ส่วนการฝึกสอนโมเดล YOLO สำหรับตรวจจับชิ้นส่วนอิเล็กทรอนิกส์ และ (2) ส่วนโปรแกรมตรวจสอบที่เปรียบเทียบผลการตรวจจับกับจุดอ้างอิง (Reference Points) เพื่อตัดสินผล PASS/FAIL

*(รูปที่ 3.1 โฟลวชาร์ตภาพรวมการทำงานของระบบ)*

ขั้นตอนการทำงาน:
1) เตรียมชุดข้อมูลภาพ PCB พร้อม Label
2) แบ่งข้อมูลเป็นชุดฝึกสอน (Training) และชุดตรวจสอบ (Validation)
3) ฝึกสอนโมเดล YOLO
4) นำโมเดลมาใช้ในโปรแกรมตรวจสอบ
5) รับภาพ PCB แล้วตรวจจับชิ้นส่วน
6) เปรียบเทียบกับจุดอ้างอิง แล้วตัดสินผล PASS/FAIL
7) บันทึกผลลงไฟล์ CSV

## 3.6 อัลกอริทึมการตรวจสอบแผ่น PCB

### 3.6.1 หลักการทำงาน

อัลกอริทึมใช้วิธี Greedy Nearest-Neighbor Matching เปรียบเทียบ Detection กับ Reference Point จับคู่ที่ใกล้ที่สุดทีละจุด แล้วตัดสินเป็น 4 สถานะ:
1) **OK** — ตรงตำแหน่งและตรงประเภท
2) **WRONG** — ตรงตำแหน่งแต่ประเภทผิด
3) **MISSING** — ไม่พบชิ้นส่วนที่ตำแหน่งอ้างอิง
4) **EXTRA** — ตรวจจับได้ชิ้นส่วนที่ไม่มีในจุดอ้างอิง

*(รูปที่ 3.9 โฟลวชาร์ตอัลกอริทึมการตรวจสอบ)*

### 3.6.2 ฟังก์ชันเปรียบเทียบผลการตรวจจับ

```python
def evaluate_inspection(reference_points, detections, match_dist, fail_on_extra):
    unmatched_det_indexes = set(range(len(detections)))
    reference_eval = []
    missing, wrong = [], []
    ok_count = 0

    for ref in reference_points:
        best_idx, best_dist = None, float("inf")
        for idx in unmatched_det_indexes:
            det = detections[idx]
            dist = math.hypot(ref["x"] - det["x"], ref["y"] - det["y"])
            if dist < best_dist:
                best_dist, best_idx = dist, idx

        if best_idx is not None and best_dist <= match_dist:
            det = detections[best_idx]
            unmatched_det_indexes.remove(best_idx)
            if det["label"] == ref["label"]:
                ok_count += 1
                reference_eval.append({"ref": ref, "det": det, "status": "OK"})
            else:
                wrong.append({"ref": ref, "det": det})
                reference_eval.append({"ref": ref, "det": det, "status": "WRONG"})
        else:
            missing.append(ref)
            reference_eval.append({"ref": ref, "det": None, "status": "MISSING"})

    extra = [detections[idx] for idx in sorted(unmatched_det_indexes)]
```

*(รูปที่ 3.10 คำสั่งเปรียบเทียบผลการตรวจจับกับจุดอ้างอิง)*

จากรูปที่ 3.10 เป็นคำสั่งเปรียบเทียบผลการตรวจจับกับจุดอ้างอิง โดยวนลูปแต่ละจุดอ้างอิงแล้วหา Detection ที่ใกล้ที่สุดด้วย Euclidean Distance (math.hypot) หากระยะไม่เกิน match_dist จะตรวจสอบว่าคลาสตรงกันหรือไม่ ตรง=OK ไม่ตรง=WRONG หาไม่เจอ=MISSING Detection ที่เหลือ=EXTRA

## 3.7 การรันโมเดล YOLO ในโปรแกรม

```python
def run_inference(self, image_path, record_history=False):
    conf_value = self.conf_slider.value() / 100.0
    results = self.model.predict(source=image_path, conf=conf_value, save=False, device="cpu")

    detections = []
    for result in results:
        annotated_frame = result.plot(labels=self.check_labels.isChecked())
        for box in result.boxes:
            cls_idx = int(box.cls[0])
            cls_name = str(self.model_names.get(cls_idx, cls_idx))
            x1, y1, x2, y2 = box.xyxy[0].tolist()
            cx, cy = int((x1+x2)/2), int((y1+y2)/2)
            confidence = float(box.conf[0])
            detections.append({"x": cx, "y": cy, "label": cls_name, "conf": confidence})

    inspection_result = evaluate_inspection(
        self.reference_points, detections,
        float(self.match_dist_spin.value()), self.check_fail_extra.isChecked())
    draw_reference_overlay(annotated_frame, inspection_result, self.check_labels.isChecked())

    if record_history:
        self.update_production_counters(inspection_result)
        self.append_history(image_path, inspection_result)
```

### 3.8.2 ระบบ Undo/Redo

```python
def save_state(self):
    self.undo_stack.append(copy.deepcopy(self.reference_points))
    self.redo_stack.clear()

def undo(self):
    if not self.undo_stack: return
    self.redo_stack.append(copy.deepcopy(self.reference_points))
    self.reference_points = self.undo_stack.pop()

def redo(self):
    if not self.redo_stack: return
    self.undo_stack.append(copy.deepcopy(self.reference_points))
    self.reference_points = self.redo_stack.pop()
```

*(รูปที่ 3.15 คำสั่งระบบ Undo/Redo)*

จากรูปที่ 3.15 ใช้ copy.deepcopy() เก็บ snapshot ของจุดอ้างอิงลง Stack ก่อนแก้ไขทุกครั้ง Undo ดึงจาก undo_stack Redo ดึงจาก redo_stack

```python
def append_history(self, image_path, inspection_result):
    timestamp = dt.datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    row = {
        "time": timestamp,
        "image": os.path.basename(image_path),
        "verdict": inspection_result["verdict"],
        "ok": inspection_result["ok"],
        "missing": len(inspection_result["missing"]),
        "wrong": len(inspection_result["wrong"]),
        "extra": len(inspection_result["extra"]),
        "station_id": self.station_id_input.text().strip(),
        "operator": self.operator_input.text().strip(),
    }
    self.history_rows.insert(0, row)
    if self.check_auto_log.isChecked():
        self.append_to_log_file(row)

def append_to_log_file(self, row):
    write_header = not os.path.exists(self.default_log_path)
    with open(self.default_log_path, "a", newline="", encoding="utf-8") as handle:
        writer = csv.writer(handle)
        if write_header:
            writer.writerow(["time","station_id","operator","image","verdict",
                             "ok","missing","wrong","extra"])
        writer.writerow([row["time"], row["station_id"], row["operator"],
                         row["image"], row["verdict"], row["ok"], row["missing"],
                         row["wrong"], row["extra"]])
```
